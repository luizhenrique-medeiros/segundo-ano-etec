    private void vapo(){
    
        int tot = Integer.parseInt(valor.getText());
        
        
    for(int cont=1; cont<=10; cont++){
        
        if(n==cont && tot>=preco[cont-1]){
            
            
        
            ImageIcon imagem12 = new  ImageIcon(salgado.class.getResource(ima[cont]));
Image imag12 = imagem12.getImage().getScaledInstance(escolha.getWidth(), escolha.getHeight(), Image.SCALE_DEFAULT);  
escolha.setIcon(new ImageIcon(imag12));

salga[cont-1].setVisible(false);

troco=tot-preco[cont-1];

tro.setText("Troco: R$"+troco);

comprar.setVisible(true);

lala=2;
        }else if(n==cont && tot<preco[cont-1]){
        
        tro.setText("Isso não é suficiente!!");
        comprar.setVisible(true);
        }
        
        if(lala==2){
        break;
        }
        }
